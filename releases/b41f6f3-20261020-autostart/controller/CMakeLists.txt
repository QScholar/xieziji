cmake_minimum_required(VERSION 3.16)
project(plotter_controller VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(plotter_control src/controller.cpp src/serial.cpp)
target_include_directories(plotter_control PUBLIC include third_party)
target_link_libraries(plotter_control PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(plotter_control PUBLIC /utf-8 /W4)
else()
  target_compile_options(plotter_control PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(plotter-controller src/main.cpp)
target_link_libraries(plotter-controller PRIVATE plotter_control)
if(WIN32)
  target_link_libraries(plotter-controller PRIVATE ws2_32 shell32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(controller-tests tests/test_controller.cpp)
  target_link_libraries(controller-tests PRIVATE plotter_control)
  add_test(NAME controller-core COMMAND controller-tests)
endif()
